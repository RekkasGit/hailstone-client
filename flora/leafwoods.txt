# Hailstone's own flora rules (assets/zone_flora.h, docs/graphics.md "Zone flora"). The leafy woods (the Feerrott):
# the shared grass with the revamped Feerrott's leafy bushes and ferns (feerrott2.eqg).
donor commonlands.eqg
donor feerrott2.eqg
max_slope 45
scale 0.9 1.3
include bare
# Ground named as grass (jungle floors too): its olive and dark grass as well, only its paths and bare patches not.
ground *jung* 12 +greenish grassblades:4 grass2:3 treeshade:2 feerrott2/grass:0.5 feerrott2/branch:0.4 feerrott2/flower:0.4
ground *grass* 12 +greenish grassblades:4 grass2:3 treeshade:2 feerrott2/grass:0.5 feerrott2/branch:0.4 feerrott2/flower:0.4
ground *gras* 12 +greenish grassblades:4 grass2:3 treeshade:2 feerrott2/grass:0.5 feerrott2/branch:0.4 feerrott2/flower:0.4
include stone
# Anything else that is clearly green.
ground * 12 +green grassblades:4 grass2:3 treeshade:2 feerrott2/grass:0.5 feerrott2/branch:0.4 feerrott2/flower:0.4
